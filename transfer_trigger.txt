CREATE OR REPLACE PROCEDURE PROC_ECF_TRANSFER(
  G_NEW_TRANS_ITEM_CODE VARCHAR2,
 G_NEW_TRANS_ITEM_DESC VARCHAR2,
 G_NEW_TRANS_TO_STORE VARCHAR2,
 G_NEW_TRANS_QTY NUMBER) IS
M_TEMP                  VARCHAR2(60) ;
M_CNT NUMBER;
CURSOR C1 IS
       SELECT 'X'
       FROM   ECF_STOCK_MASTER
       WHERE  STK_ITEM_NAME     = G_NEW_TRANS_ITEM_DESC
       AND    STK_STORE_LOC= G_NEW_TRANS_TO_STORE;
CURSOR C2 IS
 SELECT MAX(TO_NUMBER(STK_PART_NO))+1 FROM ECF_STOCK_MASTER;

BEGIN
   IF C2%ISOPEN THEN
      CLOSE C2;
  END IF;
  OPEN C2;
   FETCH C2 INTO M_CNT;
  IF C2%NOTFOUND THEN
       M_CNT:=1;
  END IF;
  CLOSE C2;
   IF C1%ISOPEN THEN
      CLOSE C1;
   END IF ;
   OPEN C1 ;
   FETCH C1 INTO M_TEMP ;
   IF C1%NOTFOUND THEN
      INSERT INTO ECF_STOCK_MASTER(STK_PKID,
STK_SR_NO,
STK_PART_NO,
STK_ITEM_NAME,
STK_DESC,
STK_UNITS,
STK_BIN_NO,
STK_REMARKS,
STK_INV_TYPE,
STK_BOOK_QTY,
STK_PH_QTY,
STK_ITEM_GROUP,
STK_MIN_LEVEL,
STK_MAX_LEVEL,
STK_QTY_STORE,
STK_QTY_ORDER,
STK_STORE_CODE,
STK_STORE_LOC,
STK_CR_DT,
STK_ITEM_BL_NAME,
STK_FRZ_YN,
STK_COST_PRICE,
STK_SELL_PRICE,
STK_REORDER_LEVEL)  SELECT ECF_STOCK_MASTER_STK_PKID.NEXTVAL,
STK_SR_NO,
NVL(M_CNT,1),
STK_ITEM_NAME,
STK_DESC,
STK_UNITS,
STK_BIN_NO,
STK_REMARKS,
STK_INV_TYPE,
STK_BOOK_QTY,
STK_PH_QTY,
STK_ITEM_GROUP,
STK_MIN_LEVEL,
STK_MAX_LEVEL,
0,
STK_QTY_ORDER,
STK_STORE_CODE,
G_NEW_TRANS_TO_STORE,
SYSDATE,
STK_ITEM_BL_NAME,
STK_FRZ_YN,
STK_COST_PRICE,
STK_SELL_PRICE,
STK_REORDER_LEVEL FROM ECF_STOCK_MASTER WHERE  STK_ITEM_NAME    = G_NEW_TRANS_ITEM_DESC;


   END IF ;
   CLOSE C1 ;

END;
/
