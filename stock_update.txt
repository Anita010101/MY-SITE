procedure stock_UPDATE(FM_DATE DATE,UPTO_DATE DATE,UID VARCHAR2) AS
CURSOR C_GOODS_REC IS
 SELECT GRV_NO,GRV_ITEM_CODE,SUM(GRV_REC_QTY) GRV_REC_QTY FROM ECF_GOODS_REC WHERE GRV_DATE BETWEEN FM_DATE AND UPTO_DATE AND NVL(GRV_PROC_FLAG,'N')='N'
  GROUP BY GRV_ITEM_CODE,GRV_NO;
CURSOR C_GOODS_ISS IS
 SELECT GIV_NO,GIV_ITEM_CODE,SUM(GIV_ISS_QTY) GIV_ISS_QTY FROM ECF_GOODS_ISSUE WHERE GIV_DATE BETWEEN FM_DATE AND UPTO_DATE AND NVL(GIV_PROC_FLAG,'N')='N'
  GROUP BY GIV_ITEM_CODE,GIV_NO;
CURSOR C_GOODS_RETN IS
 SELECT RV_NO,RV_ITEM_CODE,SUM(RV_QTY) RV_QTY FROM ECF_GOODS_RETURN WHERE RV_DATE BETWEEN FM_DATE AND UPTO_DATE AND NVL(RV_PROC_FLAG,'N')='N'
  GROUP BY RV_ITEM_CODE,RV_NO;

CURSOR C_GOODS_TRANS IS
 SELECT TRANS_NO,TRANS_ITEM_CODE,TRANS_ITEM_DESC,SUM(TRANS_QTY) TRANS_QTY,TRANS_FM_STORE,TRANS_TO_STORE FROM ECF_TRANSFER WHERE TRANS_DATE BETWEEN FM_DATE AND UPTO_DATE AND NVL(TRANS_PROC_FLAG,'N')='N'
  GROUP BY TRANS_ITEM_CODE,TRANS_ITEM_DESC,TRANS_FM_STORE,TRANS_TO_STORE,TRANS_NO;

CURSOR C_PURCHASE_ORD IS
	   SELECT PO_NO,PO_ITEM_CODE,SUM(PO_QTY) PO_QTY FROM ECF_PURCHASE_ORDER WHERE PO_DATE BETWEEN FM_DATE AND UPTO_DATE AND NVL(PO_PROC_FLAG,'N') = 'N'
	   		 GROUP BY PO_ITEM_CODE,PO_NO;
BEGIN
 FOR REC IN C_GOODS_REC
  LOOP
  UPDATE ECF_STOCK_MASTER SET STK_QTY_STORE = NVL(STK_QTY_STORE,0) + NVL(REC.GRV_REC_QTY,0) WHERE STK_PART_NO = REC.GRV_ITEM_CODE;
  UPDATE ECF_GOODS_REC SET GRV_PROC_FLAG='Y' WHERE GRV_NO = REC.GRV_NO AND GRV_DATE BETWEEN FM_DATE AND UPTO_DATE AND GRV_ITEM_CODE = REC.GRV_ITEM_CODE;
  INSERT INTO ECF_STOCK_UPDATE_LOG VALUES('GRV',REC.GRV_NO,REC.GRV_ITEM_CODE,SYSDATE,UID);
 END LOOP;

 FOR REC1 IN C_GOODS_ISS
  LOOP
  UPDATE ECF_STOCK_MASTER SET STK_QTY_STORE = NVL(STK_QTY_STORE,0) - NVL(REC1.GIV_ISS_QTY,0) WHERE STK_PART_NO = REC1.GIV_ITEM_CODE;
  UPDATE ECF_GOODS_ISSUE SET GIV_PROC_FLAG='Y' WHERE GIV_NO = REC1.GIV_NO AND GIV_DATE BETWEEN FM_DATE AND UPTO_DATE AND GIV_ITEM_CODE = REC1.GIV_ITEM_CODE;
  INSERT INTO ECF_STOCK_UPDATE_LOG VALUES('GIV',REC1.GIV_NO,REC1.GIV_ITEM_CODE,SYSDATE,UID);
 END LOOP;

 FOR REC2 IN C_GOODS_RETN
  LOOP
  UPDATE ECF_STOCK_MASTER SET STK_QTY_STORE = NVL(STK_QTY_STORE,0) + NVL(REC2.RV_QTY,0) WHERE STK_PART_NO = REC2.RV_ITEM_CODE;
  UPDATE ECF_GOODS_RETURN SET RV_PROC_FLAG='Y' WHERE RV_NO = REC2.RV_NO AND RV_DATE BETWEEN FM_DATE AND UPTO_DATE AND RV_ITEM_CODE = REC2.RV_ITEM_CODE;
  INSERT INTO ECF_STOCK_UPDATE_LOG VALUES('RV',REC2.RV_NO,REC2.RV_ITEM_CODE,SYSDATE,UID);
 END LOOP;

 FOR REC3 IN C_GOODS_TRANS
  LOOP
  UPDATE ECF_STOCK_MASTER SET STK_QTY_STORE = NVL(STK_QTY_STORE,0) + NVL(REC3.TRANS_QTY,0) WHERE STK_ITEM_NAME = REC3.TRANS_ITEM_DESC AND STK_STORE_LOC = REC3.TRANS_TO_STORE;
  UPDATE ECF_STOCK_MASTER SET STK_QTY_STORE = NVL(STK_QTY_STORE,0) - NVL(REC3.TRANS_QTY,0) WHERE STK_PART_NO = REC3.TRANS_ITEM_CODE AND STK_STORE_LOC = REC3.TRANS_FM_STORE;
  UPDATE ECF_TRANSFER SET TRANS_PROC_FLAG='Y' WHERE TRANS_NO = REC3.TRANS_NO AND TRANS_DATE BETWEEN FM_DATE AND UPTO_DATE AND TRANS_ITEM_CODE = REC3.TRANS_ITEM_CODE;
  INSERT INTO ECF_STOCK_UPDATE_LOG VALUES('TRANS',REC3.TRANS_NO,REC3.TRANS_ITEM_CODE,SYSDATE,UID);
 END LOOP;

 FOR REC4 IN C_PURCHASE_ORD
  LOOP
  UPDATE ECF_STOCK_MASTER SET STK_QTY_ORDER = NVL(STK_QTY_ORDER,0) + NVL(REC4.PO_QTY,0) WHERE STK_PART_NO = REC4.PO_ITEM_CODE;
  UPDATE ECF_PURCHASE_ORDER SET PO_PROC_FLAG='Y' WHERE PO_NO = REC4.PO_NO AND PO_DATE BETWEEN FM_DATE AND UPTO_DATE AND PO_ITEM_CODE = REC4.PO_ITEM_CODE;
  INSERT INTO ECF_STOCK_UPDATE_LOG VALUES('PO',REC4.PO_NO,REC4.PO_ITEM_CODE,SYSDATE,UID);
  END LOOP;

 COMMIT;

END;
