TRIGGER ECFINV.TRG_ECF_TRANSFER
AFTER INSERT OR UPDATE
ON ECFINV.ECF_TRANSFER 
FOR EACH ROW
DECLARE
M_TRG_MODE     VARCHAR2(1) ;
BEGIN
    IF INSERTING THEN
       M_TRG_MODE := 'I';
    ELSIF UPDATING THEN
       M_TRG_MODE := 'U';
    END IF;
    PROC_ECF_TRANSFER(
  :NEW.TRANS_ITEM_CODE,
  :NEW.TRANS_ITEM_DESC,
  :NEW.TRANS_FM_STORE,
  :NEW.TRANS_TO_STORE,
  :NEW.TRANS_QTY);
END;
